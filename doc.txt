---
- name: Deploy a Modern Web Application
  hosts: localhost
  connection: local
  become: true

  tasks:

    - name: Install Apache web server
      ansible.builtin.apt:
        name: apache2
        state: present
        update_cache: true

    - name: Start and enable Apache
      ansible.builtin.service:
        name: apache2
        state: started
        enabled: true

    - name: Deploy the modern web application
      ansible.builtin.copy:
        dest: /var/www/html/index.html
        content: |
          <!DOCTYPE html>
          <html>
          <head>
              <title>DeployFlow</title>
          </head>
          <body>
              <h1>Welcome to DeployFlow!</h1>
              <p>Web application deployed successfully using Ansible.</p>
          </body>
          </html>
        mode: "0644"

    - name: Verify the web application
      ansible.builtin.uri:
        url: http://127.0.0.1/
        status_code: 200
      register: web_result

    - name: Display deployment status
      ansible.builtin.debug:
        msg: "DeployFlow web application deployed successfully!"
ansible-playbook -i 'localhost,' -c local webapp.yml --ask-become-pass

sudo apt install ansible
ansible --version

sudo mkdir -p /etc/ansible
cd /etc/ansible

sudo ansible-galaxy init prerequisites
sudo ansible-galaxy init PostgreSQL
sudo ansible-galaxy init apache

cd /etc/ansible/prerequisites/tasks
sudo nano main.yml
# tasks file for prerequisites
- name: Install git
  apt:
    name: git
    state: present
    update_cache: yes


cd /etc/ansible/PostgreSQL/tasks
sudo nano main.yml
# tasks file for PostgreSQL

- name: Install PostgreSQL
  apt:
    name: postgresql
    state: present
    update_cache: yes

- name: Start PostgreSQL service
  systemd:
    name: postgresql
    state: started
    enabled: yes


cd /etc/ansible/apache/tasks
sudo nano main.yml
# tasks file for apache

- name: install Apache web server
  apt:
    name: apache2
    state: present
    update_cache: yes

cd /etc/ansible
sudo nano stack.yml
- hosts: localhost
  become: true
  roles:
    - prerequisites
    - PostgreSQL
    - apache

sudo ansible-playbook stack.yml
